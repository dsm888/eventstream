import org.apache.avro.io.Decoder;
import org.apache.avro.io.DecoderFactory;
import org.apache.avro.specific.SpecificDatumReader;
import org.apache.avro.generic.GenericRecord;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.stereotype.Service;

import java.io.ByteArrayInputStream;
import java.io.IOException;

@Service
public class KafkaAvroConsumer {

    @KafkaListener(topics = "my-avro-topic")
    public void listen(ConsumerRecord<String, byte[]> record) throws IOException {
        byte[] avroData = record.value();

        // Define your Avro schema (replace with your actual schema)
        String avroSchema = "{\"type\":\"record\",\"name\":\"User\",\"fields\":[{\"name\":\"name\",\"type\":\"string\"},{\"name\":\"age\",\"type\":\"int\"}]}";

        // Deserialize the Avro message
        GenericRecord avroMessage = deserializeAvro(avroData, avroSchema);
        System.out.println("Received Avro message: " + avroMessage);
    }

    private GenericRecord deserializeAvro(byte[] avroData, String avroSchema) throws IOException {
        Schema schema = new Schema.Parser().parse(avroSchema);
        SpecificDatumReader<GenericRecord> reader = new SpecificDatumReader<>(schema);
        Decoder decoder = DecoderFactory.get().binaryDecoder(new ByteArrayInputStream(avroData), null);
        return reader.read(null, decoder);
    }
}

spring:
  kafka:
    bootstrap-servers: localhost:9092
    consumer:
      group-id: my-group
      auto-offset-reset: earliest
      key-deserializer: org.apache.kafka.common.serialization.StringDeserializer
      value-deserializer: org.apache.kafka.common.serialization.ByteArrayDeserializer
